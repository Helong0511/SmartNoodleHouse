# SmartNoodleHouse 生产部署 — Ubuntu 22.04.5
# ═══════════════════════════════════════════════════════════════
# 【备案】面向国内正式营业：须中国大陆服务器 + 域名 ICP 备案 + 企业/个体户主体。
#       详见 README.md「国内上线：服务器与 ICP 备案」。
# ═══════════════════════════════════════════════════════════════
# 与 Windows 本地开发的区别见下表：
#
#   项目          本地 Windows 11              生产 Ubuntu 22.04.5
#   ─────────────────────────────────────────────────────────────
#   环境文件      .env.development.example     .env.production.example
#   APP_ENV       development                  production
#   后端监听      0.0.0.0:8000（真机）或       127.0.0.1:8000（Nginx 反代）
#                 127.0.0.1:8000（模拟器）
#   对外访问      http://127.0.0.1 / 局域网 IP   https://your-domain.com
#   小程序 API    config/env.js DEV_*          config/env.js PROD_API_BASE
#   Mock 开关     ALLOW_*_MOCK=true            全部 false
#   日志          JSON_LOGS=false              JSON_LOGS=true
#
# ═══════════════════════════════════════════════════════════════

# ── 1. 环境配置 ──
cd /path/to/SmartNoodleHouse
cp .env.production.example .env
nano .env
# 必改：JWT_SECRET、DATABASE_URL、域名、微信/支付宝密钥

# ── 2. 后端依赖与迁移 ──
cd backend
python3 -m venv .venv
source .venv/bin/activate
pip install -r requirements.txt
alembic upgrade head

# ── 3. 前端构建 ──
cd ../frontend
npm ci
npm run build
# 静态文件在 frontend/dist

# ── 4. systemd（参考 systemd/smart-noodle.service、systemd/安装说明.txt）──
#   sed -i 's|REPLACE_DEPLOY_ROOT|/var/www/smart-noodle|g' systemd/smart-noodle.service
#   sudo cp systemd/smart-noodle.service /etc/systemd/system/
#   sudo systemctl daemon-reload && sudo systemctl enable --now smart-noodle

# ── 5. Nginx（参考 nginx/smart-noodle.conf、nginx/安装说明.txt）──
#   sed -i 's|REPLACE_DOMAIN|your-domain.com|g' nginx/smart-noodle.conf
#   sed -i 's|REPLACE_WEB_ROOT|/var/www/smart-noodle/frontend/dist|g' nginx/smart-noodle.conf
#   sed -i 's|REPLACE_UPLOADS_DIR|/var/www/smart-noodle/uploads|g' nginx/smart-noodle.conf
#   sudo cp nginx/snippets/*.conf /etc/nginx/snippets/
#   sudo cp nginx/smart-noodle.conf /etc/nginx/sites-available/
#   sudo ln -sf /etc/nginx/sites-available/smart-noodle.conf /etc/nginx/sites-enabled/
#   sudo certbot --nginx -d your-domain.com
sudo nginx -t && sudo systemctl reload nginx

# ── 6. 小程序发布前 ──
#   项目根 .env：
#     密钥：python scripts/gen_secrets.py（JWT/FACE/SECRET/DISPLAY_SCREEN_TOKEN）
#     MINI_PROGRAM_ENV_VERSION=release
#     H5_PUBLIC_BASE_URL=https://your-domain.com
#     ALIPAY_AES_KEY=（支付宝手机号解密，必填）
#     ENABLE_WX_PRIVACY_APIS=true
#     ENABLE_ALIPAY_PRIVACY_APIS=true
#     PROCESS_ROLE=auto
#   厨房打印机（芯烨 XP-80T WiFi = 局域网，不是云打印）：
#     XPYUN_ENABLED=true、PRINTER_MODE=local、PRINTER_HOST=打印机IP、PRINTER_PORT=9100
#     ALLOW_PRINT_MOCK=false
#   miniprogram-wechat/config/env.js：
#     PROD_API_BASE = https://your-domain.com/api
#   miniprogram-alipay/config/env.js：
#     PROD_API_BASE = https://your-domain.com/api
#   （隐私开关已迁至根 .env，由 GET /api/miniprogram/config 下发）
#   微信 project.config.json → setting.urlCheck = true
#   微信/支付宝公众平台 → request 合法域名为 your-domain.com
#   详见 miniprogram-*/config/环境说明.txt

# ── 7. 健康检查 ──
curl -s http://127.0.0.1:8000/health
# 或：curl -s http://127.0.0.1:8000/api/ready
curl -sI https://your-domain.com/

# ── 8. 备份与恢复演练（P1-R2）──
#   每日：python3 scripts/备份数据库.py --keep-days 14
#   季度隔离库：export DRILL_DATABASE_URL=...; python3 scripts/备份恢复演练.py --restore-drill
#   恢复（危险）：python3 scripts/恢复数据库.py backups/xxx.sql.gz --yes
#   人工阻塞项：python3 scripts/上线前运维检查清单.py

# ── 9. 大表在线 DDL（P2-R2）──
#   小表 / MySQL 8 支持的即时 DDL：优先 alembic（information_schema 幂等）。
#   超大表（orders/payments/operation_logs 等）变更索引或加列：
#     - pt-online-schema-change --alter 'ADD INDEX ...' D=smart_noodle_house,t=orders --execute
#     - 或 gh-ost --alter '...' --database=... --table=orders --execute
#   变更前在从库演练；业务低峰执行；保留回滚 SQL。

# ── 10. 扩展预案（P2-R6）──
#   分片键评估：orders/payments 按 store_id + created_at 范围；热表保留 6–12 个月。
#   报表只读：配置只读副本 DATABASE_URL_READONLY（应用层后续接入）；写库仍主库。
#   JWT 多服务：JWT_ALGORITHM=RS256 并配置 JWT_PRIVATE_KEY/JWT_PUBLIC_KEY（PEM）。
