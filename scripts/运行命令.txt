# SmartNoodleHouse 本地启动命令（PowerShell）
# 在项目根目录 E:\SmartNoodleHouse 执行；前后端各开一个终端。

# ---------- 0. 先检查端口，占用则杀进程 ----------
foreach ($port in 8000, 5173) {
  $conns = Get-NetTCPConnection -LocalPort $port -ErrorAction SilentlyContinue
  if ($conns) {
    $pids = $conns | Select-Object -ExpandProperty OwningProcess -Unique | Where-Object { $_ -gt 0 }
    foreach ($procId in $pids) {
      Write-Host "端口 $port 被占用 PID=$procId，结束进程"
      Stop-Process -Id $procId -Force -ErrorAction SilentlyContinue
    }
  } else {
    Write-Host "端口 $port 空闲"
  }
}
# 残留 uvicorn（--reload 父子进程）
Get-CimInstance Win32_Process -Filter "Name='python.exe'" | Where-Object { $_.CommandLine -match 'uvicorn' } | ForEach-Object {
  Write-Host "结束 uvicorn PID=$($_.ProcessId)"
  Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue
}
Start-Sleep -Seconds 1

# ---------- 后端 ----------
cd E:\SmartNoodleHouse\backend
python -m uvicorn app.main:app --host 127.0.0.1 --port 8000 --reload

uvicorn app.main:app --reload --host 0.0.0.0 --port 8000

# 地址: http://127.0.0.1:8000
# 文档: http://127.0.0.1:8000/docs

# ---------- 前端（另开终端） ----------
cd E:\SmartNoodleHouse\frontend
# 首次或缺依赖时先: npm install
npm run dev -- --host 127.0.0.1 --port 5173
